// ============================================================ // OFFSET PRINT CO. — ORDER FORM BACKEND // Google Apps Script — paste this into script.google.com // ============================================================ const DRIVE_FOLDER_ID = '14T0grkT9KpFmxcen-mHa46kmkdeoBprH'; const NOTIFY_EMAIL = 'order@shwainc.com'; // ── Entry point ───────────────────────────────────────────── function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const orderId = generateOrderId(); const clientName = (payload.customer.name || 'Unknown').trim(); // 1. Create client folder inside the main Drive folder const clientFolder = getOrCreateFolder(clientName, DRIVE_FOLDER_ID); const orderFolder = clientFolder.createFolder('Order ' + orderId + ' — ' + today()); // 2. Save each file attachment into the order folder const fileLinks = []; if (payload.files && payload.files.length) { payload.files.forEach(f => { try { const bytes = Utilities.base64Decode(f.data); const blob = Utilities.newBlob(bytes, f.mimeType, f.name); const saved = orderFolder.createFile(blob); fileLinks.push({ name: f.name, url: saved.getUrl(), itemIndex: f.itemIndex }); } catch (err) { fileLinks.push({ name: f.name, url: 'Upload failed: ' + err.message, itemIndex: f.itemIndex }); } }); } // 3. Build and send the email sendOrderEmail(payload, orderId, orderFolder.getUrl(), fileLinks); return ContentService .createTextOutput(JSON.stringify({ status: 'ok', orderId: orderId })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // Allow CORS preflight function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ status: 'ready' })) .setMimeType(ContentService.MimeType.JSON); } // ── Email builder ──────────────────────────────────────────── function sendOrderEmail(payload, orderId, folderUrl, fileLinks) { const c = payload.customer; const items = payload.items || []; let total = 0; items.forEach(it => { total += (parseFloat(it.subtotal) || 0); }); // ── HTML email body ────────────────────────────────────── let itemRows = ''; items.forEach((it, i) => { const fileLink = fileLinks.find(f => f.itemIndex === i); const fileCell = fileLink ? `${fileLink.name}` : 'No file'; itemRows += ` ${i + 1} ${it.name || '—'}
${fileCell} ${it.h || '—'}" ${it.w || '—'}" ${it.qty || '—'} ${it.cut || '—'} ${it.material || '—'} ${it.finish || '—'} $${parseFloat(it.unitPrice || 0).toFixed(2)} $${parseFloat(it.subtotal || 0).toFixed(2)} `; }); const html = `
OFFSET PRINT CO.
New order received — ${today()}
Customer
${c.name}
${c.email}
${c.phone || ''}
${c.address || ''}
Order info
Order ID${orderId}
Shipping${c.shipping || '—'}
Priority${c.priority || '—'}
Drive folderView all files
Order items
${itemRows}
# Design H W Qty Cut Material Finish Unit Subtotal
Total order estimate
$${total.toFixed(2)}
Fee and tax not included
${payload.notes ? `
Notes
${payload.notes}
` : ''}
Offset Print Co. order system — auto-generated
`; GmailApp.sendEmail( NOTIFY_EMAIL, `[New Order] ${c.name} — ${orderId} — $${total.toFixed(2)}`, `New order from ${c.name} (${c.email}). Order ID: ${orderId}. Total: $${total.toFixed(2)}. View files: ${folderUrl}`, { htmlBody: html, name: 'Offset Print Co. Orders' } ); } // ── Helpers ────────────────────────────────────────────────── function getOrCreateFolder(name, parentId) { const parent = DriveApp.getFolderById(parentId); const existing = parent.getFoldersByName(name); return existing.hasNext() ? existing.next() : parent.createFolder(name); } function generateOrderId() { const now = new Date(); return 'OPC-' + now.getFullYear().toString().slice(2) + String(now.getMonth() + 1).padStart(2, '0') + String(now.getDate()).padStart(2, '0') + '-' + Math.random().toString(36).slice(2, 6).toUpperCase(); } function today() { return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'MMMM d, yyyy'); }
